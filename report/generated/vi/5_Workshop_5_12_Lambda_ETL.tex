\hypertarget{cost-optimization}{%
\subsubsection{Cost Optimization}}

\begin{itemize}
\tightlist
\item
  \textbf{Bedrock Titan Embed v2}: \textasciitilde\$0.0001 per 1K tokens
\item
  50 articles × 10 chunks × 500 tokens = 250K tokens/day =
  \textasciitilde\$0.025/day = \textasciitilde\$0.75/month
\item
  Lambda: 15 min × 1 GB × 1/day = negligible
\item
  \textbf{Total ETL cost}: \textasciitilde\$1-2/month
\end{itemize}

\hypertarget{performance-tips}{%
\subsubsection{Performance Tips}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Connection pooling}: Use RDS Proxy for high-frequency
  invocations
\item
  \textbf{Batch embeddings}: Process chunks in batches (though Bedrock
  is sequential)
\item
  \textbf{Retry logic}: Exponential backoff for Bedrock throttling
\item
  \textbf{Monitor HNSW index}: \texttt{ef\_search} parameter at query
  time
\end{enumerate}
