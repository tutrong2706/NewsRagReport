\subsubsection{Kết quả đạt được tuần
6:}}

\begin{itemize}
\item
  Hoàn thành \texttt{main.tf} (371 dòng) quản lý toàn bộ infrastructure:

  \begin{itemize}
  \tightlist
  \item
    \textbf{VPC}: 1 VPC, 2 public subnets, Internet Gateway, Route Table
  \item
    \textbf{Security Groups}: ECS (egress all), RDS (ingress 5432 chỉ từ
    ECS SG)
  \item
    \textbf{RDS}: Aurora PostgreSQL 15.4, instance
    \texttt{db.t4g.medium}, skip final snapshot
  \item
    \textbf{ECR}: Repository \texttt{newsrag-api}
  \item
    \textbf{ECS}: Cluster + 3 Task Definitions (crawler, etl, vectorize)
  \item
    \textbf{IAM}: Execution role (ECS Task Execution Policy) + Task role
  \item
    \textbf{CloudWatch}: Log group \texttt{/ecs/newsrag-project},
    retention 7 ngày
  \item
    \textbf{EventBridge}: 3 scheduled rules
  \end{itemize}
\item
  Cấu hình EventBridge Scheduler: \textbar{} Tên rule \textbar{} Cron
  expression \textbar{} Thời gian (UTC) \textbar{} Target Task
  \textbar{}
  \textbar------------------------\textbar----------------------\textbar-----------------\textbar-------------------\textbar{}
  \textbar{} \texttt{newsrag-crawler-rule} \textbar{}
  \texttt{cron(0\ 1\ *\ *\ ?\ *)} \textbar{} 01:00 \textbar{}
  newsrag-crawler \textbar{} \textbar{} \texttt{newsrag-etl-rule}
  \textbar{} \texttt{cron(0\ 2\ *\ *\ ?\ *)} \textbar{} 02:00 \textbar{}
  newsrag-etl \textbar{} \textbar{}
  \texttt{newsrag-vectorize-rule}\textbar{}
  \texttt{cron(0\ 3\ *\ *\ ?\ *)} \textbar{} 03:00 \textbar{}
  newsrag-vectorize \textbar{}
\item
  Environment variables được quản lý qua \texttt{locals.common\_env}:

  \begin{itemize}
  \tightlist
  \item
    DB credentials \(\rightarrow\) tự động lấy RDS endpoint
  \item
    Qdrant, Kafka, Embedding, LLM configs
  \item
    Sensitive values (passwords, API keys) \(\rightarrow\) Terraform
    variables (type=string, sensitive=true)
  \end{itemize}
\item
  Outputs: \texttt{rds\_endpoint}, \texttt{ecr\_repository\_url},
  \texttt{ecs\_cluster\_name}
\item
  \texttt{terraform\ apply} thành công, tất cả tài nguyên được tạo trên
  AWS
\end{itemize}
