\hypertarget{production-hardening}{%
\subsection{Production Hardening}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Use RDS Proxy} for connection pooling
\item
  \textbf{Secrets Manager} for DB credentials (not env vars)
\item
  \textbf{Enable X-Ray tracing} for distributed tracing
\item
  \textbf{Set up DLQ} with CloudWatch alarm
\item
  \textbf{Configure retry policy} with exponential backoff
\end{enumerate}
