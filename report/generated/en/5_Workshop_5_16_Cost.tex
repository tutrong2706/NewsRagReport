\hypertarget{optimization-strategies}{%
\subsection{Optimization Strategies}}

\hypertarget{aurora-serverless-v2}{%
\subsubsection{1. Aurora Serverless v2}}

\emph{(Mã nguồn chi tiết đã được lược bỏ để báo cáo ngắn gọn. Vui lòng
xem trong source code dự án)}

\textbf{Tips:} - Use \textbf{ACU = 0.5} minimum for dev (scales to 0
when idle) - Production: \textbf{min=1, max=16} for HA - Monitor
\texttt{ServerlessDatabaseCapacity} metric to tune - Consider
\textbf{Provisioned} with \texttt{db.t4g.medium}
(\textasciitilde\$45/mo) if steady high load

\hypertarget{lambda-right-sizing}{%
\subsubsection{2. Lambda Right-Sizing}}

\begin{Shaded}
\begin{Highlighting}[]
\CommentTok{\# Use AWS Lambda Power Tuning to find optimal memory}
\CommentTok{\# https://github.com/aws{-}lambda{-}power{-}tuning/aws{-}lambda{-}power{-}tuning}

\CommentTok{\# Typical results:}
\CommentTok{\# {-} Consumer: 512 MB (CPU scales with memory)}
\CommentTok{\# {-} ETL: 1024 MB (needs memory for pgvector ops + Bedrock)}
\CommentTok{\# {-} RAG API: 1024 MB (embedding + search + LLM)}
\end{Highlighting}
\end{Shaded}

\textbf{Cost vs Performance:} \textbar{} Memory \textbar{} CPU
\textbar{} ETL Duration \textbar{} Cost/Invocation \textbar{}
\textbar--------\textbar-----\textbar--------------\textbar-----------------\textbar{}
\textbar{} 512 MB \textbar{} 0.8 vCPU \textbar{} 180s \textbar{}
\$0.0000018 \textbar{} \textbar{} 1024 MB \textbar{} 1.7 vCPU \textbar{}
90s \textbar{} \$0.0000018 \textbar{} \textbar{} 2048 MB \textbar{} 3.5
vCPU \textbar{} 50s \textbar{} \$0.0000020 \textbar{}

\(\rightarrow\) \textbf{1024 MB is sweet spot} for ETL (2x faster, same
cost)

\hypertarget{cloudwatch-logs-retention}{%
\subsubsection{5. CloudWatch Logs
Retention}}

\begin{Shaded}
\begin{Highlighting}[]
\NormalTok{resource "aws\_cloudwatch\_log\_group" "logs" \{}
\NormalTok{  name              = "/ecs/newsrag{-}project"}
\NormalTok{  retention\_in\_days = 3  \# Reduced from 7 for dev; 14 for prod}
\NormalTok{\}}
\end{Highlighting}
\end{Shaded}

\textbf{Savings:} 50\% log cost reduction (\$0.75 vs \$1.50/month)

\hypertarget{s3-for-cold-data-optional}{%
\subsubsection{6. S3 for Cold Data
(Optional)}}

\emph{(Mã nguồn chi tiết đã được lược bỏ để báo cáo ngắn gọn. Vui lòng
xem trong source code dự án)}

Move \texttt{fact\_chunks} older than 30 days to S3, keep only recent in
pgvector.

\hypertarget{rds-proxy-for-high-concurrency}{%
\subsubsection{7. RDS Proxy (For High
Concurrency)}}

\emph{(Mã nguồn chi tiết đã được lược bỏ để báo cáo ngắn gọn. Vui lòng
xem trong source code dự án)}

\textbf{When to use:} \textgreater{} 50 concurrent Lambda connections
\textbf{Cost:} \textasciitilde\$0.015/hr per vCPU =
\textasciitilde\$11/month for db.t4g.medium equivalent \textbf{Benefit:}
Connection pooling, reduced cold start, better scalability
