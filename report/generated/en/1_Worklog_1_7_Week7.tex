\hypertarget{week-7-results}{%
\subsubsection{Week 7 Results:}}

\begin{itemize}
\item
  Successful integration test on full pipeline:

  \begin{itemize}
  \tightlist
  \item
    Crawls \textasciitilde500 articles per run from 3 newspapers
  \item
    ETL processes: clean HTML \(\rightarrow\) chunk 800 chars (overlap
    150) \(\rightarrow\) insert Star Schema
  \item
    Vectorize: embedding with \texttt{BAAI/bge-small-en-v1.5}
    \(\rightarrow\) upsert to Qdrant
  \item
    RAG API: query \(\rightarrow\) embed \(\rightarrow\) search top-k
    \(\rightarrow\) LLM generates answer
  \end{itemize}
\item
  Handled critical edge cases:

  \begin{itemize}
  \tightlist
  \item
    \textbf{Invalid author}: Detected and removed 15+ types of fake
    authors (URLs, dates, newspaper names, special chars)
  \item
    \textbf{Date parsing}: Supports 3 formats (ISO, VN dd/mm/yyyy,
    date-only) + fallback through 10 CSS selectors
  \item
    \textbf{Empty content}: Skips articles with content \textless{} 100
    characters
  \item
    \textbf{ETL author cleanup}: Removes authors \textgreater{} 40
    chars, containing http/\textbar/@
  \end{itemize}
\item
  Ragas evaluation results: \textbar{} Metric \textbar{} NewsRAG
  \textbar{} FlashRAG \textbar{}

  \textbf{Faithfulness} \textbar{} Average \textbar{} \textbf{Higher}
  \textbar{} \textbar{} \textbf{Answer Relevancy} \textbar{} Comparable

  configured to provide detailed, comprehensive answers \(\rightarrow\)
  reduces cosine similarity when Ragas reverse-translates to questions.
  \end{quote}
